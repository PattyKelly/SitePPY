# 15) Comandos rápidos para começar (copy/paste)
No Windows PowerShell (adaptar caminhos):
# 1. clonar
git clone <url-repo>
cd repo
# 2. node setup
npm install
# 3. python venv
python -m venv .venv
Set-ExecutionPolicy -ExecutionPolicy RemoteSigned -Scope CurrentUser  # se necessário
.venv\Scripts\Activate.ps1
pip install -r requirements.txt
# 4. rodar site local (Astro)
npm run dev
# 5. gerar relatorio python local
python scripts/relatorio.py --out relatorio_site.csv
# 6. testar função netlify (opcional)
# instalar netlify cli: npm i -g netlify-cli
netlify dev
